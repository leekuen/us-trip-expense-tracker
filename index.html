⚙️ 雲端同步設定

設定後，記帳會自動同步到 Google 試算表「記帳」分頁。離線時新增/編輯/刪除都會先存在手機上，恢復連線後自動補同步。

還沒設定？點我看一次性設定步驟
  1. 打開你的 Google 試算表 →上方選單「擴充功能」→「Apps Script」。
  2. 把編輯器裡原本的程式碼全部刪掉，貼上下面這段程式碼：
function doPost(e) {
  try {
    var payload = JSON.parse(e.postData.contents);
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('記帳');
    if (!sheet) throw new Error('找不到「記帳」分頁');
    ensureBookkeepingColumns_(sheet);
    var tz = SpreadsheetApp.getActiveSpreadsheet().getSpreadsheetTimeZone();
    var action = payload.action || 'create';
    var d = payload.data || {};
    var now = new Date().toISOString();

    if (action === 'create') {
      // Idempotent: if this device already successfully created this record
      // before but never received the response (flaky mobile network), retry
      // must not append a second row — look it up by the client-generated id.
      var localId = String(payload.localId || '').trim();
      if (localId) {
        var existingRow = findRowByLocalId_(sheet, localId);
        if (existingRow) return json_({ status: 'ok', row: existingRow, lastModified: d.lastModified || now });
      }
      sheet.appendRow([
        d.date || '', d.category || '', d.item || '', d.method || '',
        d.amountUSD != null ? d.amountUSD : '',
        d.amountTWD != null ? d.amountTWD : '',
        d.note || '',
        d.actualCardTWD != null ? d.actualCardTWD : '',
        d.lastModified || now,
        localId
      ]);
      return json_({ status: 'ok', row: sheet.getLastRow(), lastModified: d.lastModified || now });
    }

    if (action === 'update') {
      if (!payload.row) throw new Error('缺少 row');
      // 只更新前 8 欄，不動「實際刷卡台幣金額」以外的東西已在 App 端排除，
      // 但仍整列連同該欄一起寫回，讓電腦端已填的對帳金額不會消失。
      sheet.getRange(payload.row, 1, 1, 9).setValues([[
        d.date || '', d.category || '', d.item || '', d.method || '',
        d.amountUSD != null ? d.amountUSD : '',
        d.amountTWD != null ? d.amountTWD : '',
        d.note || '',
        d.actualCardTWD != null ? d.actualCardTWD : '',
        d.lastModified || now
      ]]);
      // Backfill the local-id column for rows created before this idempotency
      // fix existed, so a future retry of this row is also protected.
      var updLocalId = String(payload.localId || '').trim();
      if (updLocalId) sheet.getRange(payload.row, 10).setValue(updLocalId);
      return json_({ status: 'ok' });
    }

    if (action === 'delete') {
      if (!payload.row) throw new Error('缺少 row');
      sheet.getRange(payload.row, 1, 1, 8).clearContent();
      sheet.getRange(payload.row, 9).setValue(d.lastModified || now);
      return json_({ status: 'ok' });
    }

    if (action === 'pull') {
      var since = payload.since || '';
      var lastRow = sheet.getLastRow();
      var rows = [];
      if (lastRow >= 2) {
        var values = sheet.getRange(2, 1, lastRow - 1, 9).getValues();
        for (var i = 0; i < values.length; i++) {
          var row = values[i];
          var rowNum = i + 2;
          var hasData = row[0] || row[1] || row[2] || row[3] || row[4] !== '' || row[5] !== '' || row[6] || row[7] !== '';
          var lm = row[8];
          var lmStr = lm instanceof Date ? lm.toISOString() : String(lm || '').trim();
          if (!lmStr && hasData) {
            lmStr = now;
            sheet.getRange(rowNum, 9).setValue(lmStr);
          }
          if (!lmStr) continue;
          if (since && lmStr <= since) continue;
          rows.push({
            row: rowNum,
            deleted: !hasData,
            lastModified: lmStr,
            data: {
              // Sheets auto-types a "2026-09-01"-looking string into a real
              // Date cell. getValues() then hands back a JS Date object, and
              // naively JSON-serializing that renders in UTC — for a GMT+8
              // sheet that silently shifts the calendar date back a day.
              // Re-format explicitly using the spreadsheet's own timezone.
              date: normalizeDateCell_(row[0], tz), category: row[1], item: row[2], method: row[3],
              amountUSD: row[4], amountTWD: row[5], note: row[6], actualCardTWD: row[7]
            }
          });
        }
      }
      var opts = getDropdownOptions_(sheet);
      return json_({ status: 'ok', rows: rows, serverTime: now, categories: opts.categories, methods: opts.methods });
    }

    if (action === 'meta') {
      var opts2 = getDropdownOptions_(sheet);
      return json_({ status: 'ok', categories: opts2.categories, methods: opts2.methods });
    }

    if (action === 'addOption') {
      var field = payload.field; // 'category' | 'method'
      var newValue = String(payload.value || '').trim();
      if (!newValue) throw new Error('缺少要新增的值');
      var headers2 = sheet.getRange(1, 1, 1, 8).getValues()[0];
      var col = findCol_(headers2, field === 'category' ? '大項目' : '支付方式');
      if (col < 1) throw new Error('找不到欄位');
      var rule2 = sheet.getRange(2, col).getDataValidation();
      if (!rule2 || rule2.getCriteriaType() != SpreadsheetApp.DataValidationCriteria.VALUE_IN_LIST) {
        throw new Error('這個欄位的下拉選單不是簡單清單型態，請直接到試算表「資料 > 資料驗證」手動加選項');
      }
      var existing = rule2.getCriteriaValues()[0];
      if (existing.indexOf(newValue) === -1) existing.push(newValue);
      var newRule = SpreadsheetApp.newDataValidation().requireValueInList(existing, true).build();
      sheet.getRange(2, col, sheet.getMaxRows() - 1, 1).setDataValidation(newRule);
      return json_({ status: 'ok', categories: field === 'category' ? existing : undefined, methods: field === 'method' ? existing : undefined });
    }

    throw new Error('未知的 action: ' + action);
  } catch (err) {
    return json_({ status: 'error', message: err.message });
  }
}

function findCol_(headers, keyword) {
  for (var i = 0; i < headers.length; i++) {
    if (String(headers[i]).indexOf(keyword) !== -1) return i + 1;
  }
  return -1;
}

function getDropdownOptions_(sheet) {
  var headers = sheet.getRange(1, 1, 1, 8).getValues()[0];
  var catCol = findCol_(headers, '大項目');
  var methodCol = findCol_(headers, '支付方式');
  return {
    categories: catCol > 0 ? getListValidationValues_(sheet, catCol) : [],
    methods: methodCol > 0 ? getListValidationValues_(sheet, methodCol) : []
  };
}

function getListValidationValues_(sheet, col) {
  var rule = sheet.getRange(2, col).getDataValidation();
  if (!rule) return [];
  var criteria = rule.getCriteriaType();
  var values = rule.getCriteriaValues();
  if (criteria == SpreadsheetApp.DataValidationCriteria.VALUE_IN_LIST) {
    return values[0];
  }
  if (criteria == SpreadsheetApp.DataValidationCriteria.VALUE_IN_RANGE) {
    return values[0].getValues().reduce(function (a, b) { return a.concat(b); }, []).filter(String);
  }
  return [];
}

function ensureBookkeepingColumns_(sheet) {
  var h9 = sheet.getRange(1, 9).getValue();
  if (!h9) sheet.getRange(1, 9).setValue('最後修改時間（同步用，勿刪）');
  var h10 = sheet.getRange(1, 10).getValue();
  if (!h10) sheet.getRange(1, 10).setValue('本機記錄ID（同步用，勿刪）');
}

function findRowByLocalId_(sheet, localId) {
  var lastRow = sheet.getLastRow();
  if (lastRow < 2) return null;
  var ids = sheet.getRange(2, 10, lastRow - 1, 1).getValues();
  for (var i = 0; i < ids.length; i++) {
    if (String(ids[i][0]) === localId) return i + 2;
  }
  return null;
}

function normalizeDateCell_(v, tz) {
  if (v instanceof Date) return Utilities.formatDate(v, tz, 'yyyy-MM-dd');
  return v;
}

function json_(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON);
}

function doGet(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('記帳');
  return ContentService.createTextOutput(
    '✅ 記帳連動已就緒，目前列數：' + (sheet ? sheet.getLastRow() : '找不到「記帳」分頁')
  );
}

📲 加到手機主畫面

iPhone（Safari）：打開這個網址 → 點下方分享圖示 ⬆️ → 選「加入主畫面」→ 完成後桌面會有一個圖示，點開就像 App 一樣全螢幕、沒有網址列。

Android（Chrome）：打開這個網址 → 右上角選單 ⋮ → 選「新增至主畫面」或「安裝應用程式」。
